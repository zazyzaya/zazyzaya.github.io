    $nums \gets [1,...,100]$\;
    \For{$i \in nums$}{
        \If{$i \mod 3 == 0 \land i \mod 5 == 0$}{
            print(``FizzBuzz")\;
        } \ElseIf {$i \mod 3 == 0$} {
            print(``Fizz")\;
        } \ElseIf {$i \mod 5 == 0$} {
            print(``Buzz")\;
        } \Else {
            print($i$)\;
        }
    }
\end{algorithm}
